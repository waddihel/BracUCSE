//Task 6
public static Integer subtractSummation(BTNode root) {
  return subSumHelper(root.left) - subSumHelper(root.right);
}

private static Integer subSumHelper(BTNode root) {
  if (root == null) return 0;
  return (Integer) root.elem + subSumHelper(root.left) + subSumHelper(root.right);
}

//Task 7
public static Integer levelSum(BTNode root) {
  return lvlSumHelper(root, 0);
}

private static Integer lvlSumHelper(BTNode root, int lvl) {
  if (root == null) return 0;
  if (lvl % 2 == 0)
    return -(Integer) root.elem
        + lvlSumHelper(root.left, lvl + 1)
        + lvlSumHelper(root.right, lvl + 1);
  return (Integer) root.elem
      + lvlSumHelper(root.left, lvl + 1)
      + lvlSumHelper(root.right, lvl + 1);
}

//Task 8
public static void swapChild(BTNode root, int lvl, int M) {
  if (lvl == M || root == null) return;
  BTNode temp = root.left;
  root.left = root.right;
  root.right = temp;
  swapChild(root.left, lvl + 1, M);
  swapChild(root.right, lvl + 1, M);
}

//Task 4
public static Integer rangeSum(BSTNode root, Integer low, Integer high) {
  if (root == null) return 0;
  if ((Integer) root.elem >= low && (Integer) root.elem <= high)
    return (Integer) root.elem + rangeSum(root.left, low, high) + rangeSum(root.right, low, high);
  if ((Integer) root.elem < low) return rangeSum(root.right, low, high);
  return rangeSum(root.left, low, high);
}

//Task 5
public static Integer mirrorSum(BSTNode root) {
  return mSumHelp(root.left, root.right);
}

private static Integer mSumHelp(BSTNode left, BSTNode right) {
  return (left != null && right != null)
      ? ((int) left.elem
          + right.elem
          + mSumHelp(left.left, right.right)
          + mSumHelp(left.right, right.left))
      : 0;
}

//Task 6
public static Boolean isBST(BSTNode root) {
  if (root.left == null && root.right == null) return true;
  if (root.left == null) {
    if (root.elem < root.right.elem) {
      if (!isBST(root.right)) return false;
      return true;
    }
  }
  if (root.right == null) {
    if (root.elem > root.left.elem) {
      if (!isBST(root.left)) return false;
      return true;
    }
  }
  if (root.left.elem < root.elem && root.elem < root.right.elem) {
    if (!(isBST(root.left) && isBST(root.right))) return false;
    return true;
  }
  return false;
}
